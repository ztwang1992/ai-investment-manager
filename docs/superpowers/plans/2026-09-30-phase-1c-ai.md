**Goal:** 按原型 v5 实现「AI 投顾」页。
- **接入页**：服务商列表、接口地址 / 模型 / Key、**真实的测试连接**、数据授权、开始对话；

**Architecture:**
- **连接测试**：浏览器直接请求用户填的 `https://…/models`。已验证 DeepSeek 对任何来源都放行跨域，401 响应也带跨域头，所以不需要转发。请求逻辑在 `connectionTest.ts`，`fetch` 可注入，测试不走网络。

- **会话状态**：单独的 zustand store（`aiStore.ts`），和 App 主状态分开。Key 只在内存里：不进主状态、不进备份、不打日志，只发给用户填的 https 接口地址。
- **欢迎语数据**：复用计划页的 `usePlanData`。
- **留到第 6 阶段**：Key 用 WebCrypto 加密存 IndexedDB、发给模型的数据摘要、真实回复（流式）、会话存 Supabase。

export type TestStatus = 'testing' | 'ok' | 'fail';

export interface TestResult {
  status: TestStatus;
  title: string;
  message: string;
}

const fail = (message: string, title = '连接失败'): TestResult => ({ status: 'fail', title, message });

/** 发请求前的检查；返回 null 表示可以发。Key 只发给 https 地址。 */
export function precheck(i: { base: string; model: string; key: string }): TestResult | null {
  if (!i.key.trim()) return fail('请先填写 API Key');
  if (!i.base.trim()) return fail('请先填写接口地址');
  if (!i.base.trim().startsWith('https://')) return fail('接口地址要以 https:// 开头');
  if (!i.model.trim()) return fail('请先填写模型');
  return null;
}

export function testingText(base: string): TestResult {
  return { status: 'testing', title: '正在测试连通性…', message: `请求 ${base.trim().replace(/\/+$/, '')}/models` };
}

export function testResultText(outcome: TestOutcome, model: string): TestResult {
  const wanted = model.trim();
  switch (outcome.kind) {
    case 'ok':
      if (outcome.models.length > 0 && !outcome.models.includes(wanted)) {
        return fail(`连接成功，但可用模型里没有 ${wanted}。可用模型：${outcome.models.join('、')}`, '模型不可用');
      }
      return { status: 'ok', title: '连接成功', message: `延迟 ${outcome.latencyMs}ms · 可用模型：${outcome.models.join('、') || wanted}` };
    case 'http':
      if (outcome.status === 401) return fail('401 鉴权失败：API Key 无效或已过期');
      if (outcome.status === 402) return fail('402 余额不足：请到 DeepSeek 控制台充值');
      return fail(`${outcome.status} 请求失败：请检查接口地址`);
    case 'timeout':
      return fail('连接超时：请检查网络或接口地址');
    case 'network':
      return fail('无法连接：请检查网络和接口地址');
  }
}

/** 新对话的第一句（原型 welcome()）；没有偏离项时不写「最大的是」。 */
export function welcomeText(i: { model: string; total: string; offCount: number; biggest: string | null }): string {
  const status =
    i.offCount > 0 && i.biggest ? `目前 ${i.offCount} 项资产偏离目标，最大的是${i.biggest}` : '目前各项资产都在目标范围内';
  return `已连接 ${PROVIDER} · ${i.model}。我读取了你的计划、持仓和收益数据：总资产约 ${i.total}，${status}。\n想从哪里聊起？`;
}

/** 新对话用第一个问题的前 18 个字作标题。 */
export function conversationTitle(current: string, question: string): string {
  return current === NEW_TITLE ? Array.from(question).slice(0, 18).join('') : current;
}

export function busyText(web: boolean): string {
  return `AI 投顾正在分析${web ? '并查询最新市场信息' : ''}…`;
}

export function conversationMeta(c: Pick<Conversation, 'time' | 'messages'>): string {
  return `${c.time} · ${c.messages.length} 条消息`;
}

export function connectionLabel(model: string): string {
  return `${PROVIDER} · ${model} · 已连接`;
}

import { afterEach, describe, expect, it, vi } from 'vitest';
import { requestModels } from './connectionTest';

const response = (status: number, json: () => Promise<unknown>) =>
  ({ ok: status >= 200 && status < 300, status, json }) as unknown as Response;
const deps = (fetch: unknown, now: () => number = () => 0) => ({
  fetch: fetch as typeof globalThis.fetch,
  now,
  timeoutMs: 6000,
});

afterEach(() => {
  vi.useRealTimers();
});

describe('requestModels', () => {
  it('asks {base}/models with the key and reports the latency and the models', async () => {
    let clock = 1000;
    const fetch = vi.fn(async () => {
      clock += 142;
      return response(200, async () => ({ data: [{ id: 'deepseek-chat' }, { id: 'deepseek-reasoner' }] }));
    });
    const outcome = await requestModels({ base: ' https://api.deepseek.com/ ', key: 'sk-abc' }, deps(fetch, () => clock));
    expect(outcome).toEqual({ kind: 'ok', latencyMs: 142, models: ['deepseek-chat', 'deepseek-reasoner'] });
    expect(fetch).toHaveBeenCalledTimes(1);
    expect(fetch).toHaveBeenCalledWith(
      'https://api.deepseek.com/models',
      expect.objectContaining({ headers: { Authorization: 'Bearer sk-abc' } }),
    );
  });

  it('passes on HTTP errors such as 401', async () => {
    const fetch = vi.fn(async () => response(401, async () => ({ error: { message: 'Authentication Fails' } })));
    expect(await requestModels({ base: 'https://api.deepseek.com', key: 'sk-bad' }, deps(fetch))).toEqual({ kind: 'http', status: 401 });
  });

  it('copes with a success that is not JSON', async () => {
    const fetch = vi.fn(async () =>
      response(200, async () => {
        throw new SyntaxError('not json');
      }),
    );
    expect(await requestModels({ base: 'https://api.deepseek.com', key: 'sk-abc' }, deps(fetch))).toEqual({
      kind: 'ok',
      latencyMs: 0,
      models: [],
    });
  });

  it('gives up after the timeout', async () => {
    vi.useFakeTimers();

      (_url: string, init: RequestInit) =>
        new Promise<Response>((_resolve, reject) => {

```ts
import type { TestOutcome } from './types';

// 「测试连接」：请求 {接口地址}/models（OpenAI 兼容格式）。
// 浏览器直接发给服务商：DeepSeek 对任何来源都放行跨域，401 响应也带跨域头，所以不需要转发。
// Key 只放在 Authorization 头里，这里不记录、不打日志；出错时只返回类型，不带请求内容。

export interface RequestDeps {
  fetch: typeof fetch;
  now: () => number;
  timeoutMs: number;
}

export const defaultRequestDeps: RequestDeps = {
  // 调用时才取 globalThis.fetch，测试可以用 vi.stubGlobal 替换
  fetch: (input, init) => globalThis.fetch(input, init),
  now: () => performance.now(),
  timeoutMs: 6000,
};

export async function requestModels(i: { base: string; key: string }, deps: RequestDeps = defaultRequestDeps): Promise<TestOutcome> {
  const url = `${i.base.trim().replace(/\/+$/, '')}/models`;
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), deps.timeoutMs);
  const started = deps.now();
  try {
    const res = await deps.fetch(url, { headers: { Authorization: `Bearer ${i.key}` }, signal: controller.signal });
    const latencyMs = Math.round(deps.now() - started);
    if (!res.ok) return { kind: 'http', status: res.status };
    const body = (await res.json().catch(() => ({}))) as { data?: { id?: unknown }[] };
    const models = (Array.isArray(body.data) ? body.data : [])
      .map((m) => m.id)
      .filter((id): id is string => typeof id === 'string');
    return { kind: 'ok', latencyMs, models };
  } catch {
    return controller.signal.aborted ? { kind: 'timeout' } : { kind: 'network' };
  } finally {
    clearTimeout(timer);
  }
}

import type { Conversation, TestOutcome } from './types';

// AI 投顾的会话状态，和 App 的主状态分开：Key 只放在内存里，不进主状态、不进备份、不打日志，
// 只在「测试连接」时作为 Authorization 头发给用户填的 https 地址。
// 第 6 阶段：Key 用 WebCrypto 加密后存 IndexedDB，会话存 Supabase，回复改为请求 DeepSeek。

export interface AiDeps {
  now: () => Date;
  delay: (ms: number) => Promise<void>;
  newId: () => string;
  requestModels: (i: { base: string; key: string }) => Promise<TestOutcome>;

export interface AiState {
  base: string;
  model: string;
  /** 只在内存里；不进主状态、不进备份、不打日志 */
  key: string;
  consent: boolean;
  test: TestResult | null;
  connected: boolean;
  web: boolean;
  conversations: Conversation[];
  currentId: string | null;
  busy: boolean;
  setBase: (value: string) => void;
  setModel: (value: string) => void;
  setKey: (value: string) => void;

  newConversation: (welcome: string, today: string) => void;
  select: (id: string) => void;
  toggleWeb: () => void;
  ask: (question: string) => Promise<void>;
}

/** 固定回复前的停顿，让「正在分析」的状态看得见 */
const REPLY_DELAY_MS = 800;

const defaultDeps: AiDeps = {
  now: () => new Date(),
  delay: (ms) => new Promise((resolve) => setTimeout(resolve, ms)),
  newId,
  requestModels: (i) => requestModels(i),
};

export function createAiStore(deps: AiDeps = defaultDeps) {
  return create<AiState>()((set, get) => {
    // 每次改表单或重新测试都换一个编号，旧请求回来时编号对不上就丢掉
    let testSeq = 0;
    const hm = () => formatTimeHM(deps.now());
    const edit = (patch: Partial<Pick<AiState, 'base' | 'model' | 'key'>>) => {
      testSeq += 1;
      set({ ...patch, test: null });
    };
    const update = (id: string, change: (c: Conversation) => Conversation) =>
      set((s) => ({ conversations: s.conversations.map((c) => (c.id === id ? change(c) : c)) }));
    const newConversation = (welcome: string, today: string) => {
      const id = deps.newId();
      const time = hm();
      const conversation: Conversation = {
        id,
        title: NEW_TITLE,
        time: `${today.slice(5)} ${time}`,
        messages: [{ who: 'ai', text: welcome, time }],
      };
      set((s) => ({ conversations: [conversation, ...s.conversations], currentId: id }));
    };

    return {
      base: DEFAULT_BASE,
      model: DEFAULT_MODEL,
      key: '',
      consent: false,
      test: null,
      connected: false,
      web: true,
      conversations: seedConversations.map((c) => ({ ...c, messages: [...c.messages] })),
      currentId: null,
      busy: false,
      setBase: (base) => edit({ base }),
      setModel: (model) => edit({ model }),
      setKey: (key) => edit({ key }),
      toggleConsent: () => set((s) => ({ consent: !s.consent })),
      runTest: async () => {
        const form = get();
        const seq = ++testSeq;
        const problem = precheck(form);
        if (problem) {
          set({ test: problem });
          return;
        }
        set({ test: testingText(form.base) });
        const outcome = await deps.requestModels({ base: form.base, key: form.key.trim() });
        if (seq !== testSeq) return;
        set({ test: testResultText(outcome, form.model) });
      },
      connect: (welcome, today) => {
        const s = get();
        if (s.test?.status !== 'ok' || !s.consent) return;
        set({ connected: true });
        newConversation(welcome, today);
      },
      disconnect: () => {
        testSeq += 1;
        set({ connected: false, key: '', test: null });
      },
      newConversation,
      select: (id) => set({ currentId: id }),
      toggleWeb: () => set((s) => ({ web: !s.web })),
      ask: async (question) => {
        const q = question.trim();
        const { busy, currentId } = get();
        if (!q || busy || !currentId) return;
        update(currentId, (c) => ({
          ...c,
          title: conversationTitle(c.title, q),
          messages: [...c.messages, { who: 'me', text: q, time: hm() }],
        }));
        set({ busy: true });
        await deps.delay(REPLY_DELAY_MS);
        update(currentId, (c) => ({ ...c, messages: [...c.messages, { who: 'ai', text: FIXED_REPLY, time: hm() }] }));
        set({ busy: false });
      },
    };
  });
}

export const useAiStore = createAiStore();
```

import { useEffect, useRef, useState } from 'react';
import type { KeyboardEvent } from 'react';
import { PanelLeftIcon, SendIcon } from '../../app/icons';
import { useAppStore } from '../../app/store';
import { useAiStore } from './aiStore';
import { PROVIDER, SUGGESTIONS, busyText } from './aiText';

/** 对话页：频道式标题、已读取标签、联网开关、消息、推荐问题、输入框（原型 v5 第 257–283 行）。 */
export function AiChat() {
  const { conversations, currentId, busy, web, ask, toggleWeb, newConversation } = useAiStore();
  const today = useAppStore((s) => s.today);
  const welcome = useWelcome();
  const [input, setInput] = useState('');
  const listRef = useRef<HTMLDivElement>(null);
  const current = conversations.find((c) => c.id === currentId);
  const messages = current?.messages ?? [];

  // 新消息或切换对话后滚到底部
  useEffect(() => {
    const list = listRef.current;
    if (list) list.scrollTop = list.scrollHeight;
  }, [messages.length, busy, currentId]);

  const send = () => {
    if (!input.trim() || busy) return;
    void ask(input);
    setInput('');
  };
  // 输入法选字时的回车只确认候选字（Safari 里 isComposing 可能为 false，但 keyCode 是 229）
  const onKeyDown = (e: KeyboardEvent<HTMLInputElement>) => {
    if (e.key !== 'Enter' || e.nativeEvent.isComposing || e.keyCode === 229) return;
    e.preventDefault();
    send();
  };

  return (
    <section className="ai-chat">
      <header className="ai-chat-head">
        <div className="ai-chat-bar">
          <div className="ai-chat-title-wrap">
            <button type="button" className="btn btn-secondary btn-icon" aria-label="历史对话">
              <PanelLeftIcon />
            </button>
            <h1 className="ai-chat-title">
              <span className="ai-hash">#</span> {current?.title ?? '我的组合'}
            </h1>
          </div>
          <button type="button" className="btn btn-ghost ai-small" onClick={() => newConversation(welcome, today)}>
            + 新对话
          </button>
        </div>
        <div className="ai-tags">
          <span className="tag tag-accent-2">已读取 计划 · 持仓 · 收益</span>
          <span className="tag tag-neutral">{PROVIDER}</span>
          <button type="button" className={`tag ai-web ${web ? 'tag-accent' : 'tag-outline'}`} aria-pressed={web} onClick={toggleWeb}>
            {web ? '联网分析 · 开' : '联网分析 · 关'}
          </button>
        </div>
      </header>
      <div className="ai-msgs" ref={listRef}>
        {messages.map((m, i) => (
          <div key={i} className="ai-msg">
            <span className={`ai-avatar ${m.who === 'ai' ? 'is-ai' : 'is-me'}`}>{m.who === 'ai' ? 'AI' : '我'}</span>
            <div className="ai-msg-body">
              <div className="ai-msg-head">
                <span className="ai-msg-name">{m.who === 'ai' ? 'AI 投顾' : '我'}</span>
                <span className="ai-msg-time">{m.time}</span>
              </div>
              <span className="ai-msg-text">{m.text}</span>
            </div>
          </div>
        ))}
        {busy && (
          <div className="ai-busy" role="status">
            <span className="ai-busy-dot" />
            {busyText(web)}
          </div>
        )}
      </div>
      {messages.length <= 1 && !busy && (
        <div className="ai-suggests">
          {SUGGESTIONS.map((q) => (
            <button key={q} type="button" className="btn btn-secondary ai-suggest" onClick={() => void ask(q)}>
              {q}
            </button>
          ))}
        </div>
      )}
      <div className="ai-input-row">
        <input
          className="input ai-input"
          aria-label="问问你的组合"
          placeholder="问问你的组合…"
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={onKeyDown}
        />
        <button type="button" className="btn btn-primary btn-icon ai-send" aria-label="发送" onClick={send}>
          <SendIcon />
        </button>
      </div>
    </section>
  );
}

```tsx
// ConversationDrawer.tsx
import { useEffect } from 'react';
import { createPortal } from 'react-dom';
import { overlayRoot } from '../../app/Sheet';
import { useAppStore } from '../../app/store';
import { useAiStore } from './aiStore';
import { connectionLabel, conversationMeta } from './aiText';
import { useWelcome } from './useWelcome';

/** 历史对话抽屉：会话列表、新对话、更换 Key（原型 v5 第 285–294 行）。点空白处或按 Esc 关闭。 */
export function ConversationDrawer({ onClose }: { onClose: () => void }) {
  const { conversations, currentId, model, select, newConversation, disconnect } = useAiStore();
  const today = useAppStore((s) => s.today);
  const welcome = useWelcome();

  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [onClose]);

  return createPortal(
    <div className="ai-drawer-backdrop" onClick={onClose}>
      <div className="ai-drawer" role="dialog" aria-modal="true" aria-label="历史对话" onClick={(e) => e.stopPropagation()}>
        <div className="ai-drawer-head">
          <h2 className="ai-drawer-title">历史对话</h2>
          <button
            type="button"
            className="btn btn-primary ai-drawer-new"
            onClick={() => {
              newConversation(welcome, today);
              onClose();
            }}
          >
            + 新对话
          </button>
        </div>
        <div className="ai-drawer-list">
          {conversations.map((c) => (
            <button
              key={c.id}
              type="button"
              className="ai-conv"
              aria-current={c.id === currentId ? 'true' : undefined}
              onClick={() => {
                select(c.id);
                onClose();
              }}
            >
              <span className="ai-conv-title">{c.title}</span>
              <span className="ai-conv-meta">{conversationMeta(c)}</span>
            </button>
          ))}
        </div>
        <div className="ai-drawer-foot">
          <span className="ai-conn">{connectionLabel(model)}</span>
          <button
            type="button"
            className="btn btn-ghost ai-small"
            onClick={() => {
              disconnect();
              onClose();
            }}
          >
            更换 Key
          </button>
        </div>
      </div>
    </div>,
    overlayRoot(),
  );
}
```
